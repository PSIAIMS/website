Plotly

Below are plotly output and the datatable from the earlier part of the module:

f1 <- ggplot(data = adsl, aes(x=TRT01P,y=TRTDURD,fill=TRT01P)) +
  geom_boxplot() +
  scale_x_discrete(name="Treatment Group") +
  scale_y_continuous(limits=c(0,250), breaks=seq(0,250,50), expand=c(0.05,0.05), name="Treatment Duration") +
  scale_fill_manual(values = c("#0072B2", "#D55E00", "#CC79A7"), name="Treatment Group") +
  theme_bw()

ggplotly(f1)

Datatable

adslm <- adsl %>% 
         mutate_if(is.character, as.factor)

datatable(adslm,
          extensions = c('ColReorder'), 
          rownames = FALSE, filter = 'top',
          options = list(colReorder = TRUE, lengthMenu = c(18,50,100),
                         scrollX = FALSE))
